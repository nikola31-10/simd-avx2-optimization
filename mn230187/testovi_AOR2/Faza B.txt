#include "io.h"
#include "files.h"

void
__attribute__((noreturn))
__attribute__((section(".start")))
_start(void)
{
    char buffer[5];
    int fd;
    int position;
    int count;
    const char *s;

    fd = open("seektest.txt", FILE_O_RD);

    if (fd < 0) {
        s = "B2 OPEN FAIL\n";
        while (*s)
            outb(0xE9, (uint8_t)*s++);
        goto finish;
    }

    position = lseek(fd, 5, FILE_SEEK_SET);
    count = read(fd, buffer, 4);
    close(fd);

    buffer[4] = '\0';

    if (position == 5 &&
        count == 4 &&
        buffer[0] == '5' &&
        buffer[1] == '6' &&
        buffer[2] == '7' &&
        buffer[3] == '8') {

        s = "B2 PASS: ";
        while (*s)
            outb(0xE9, (uint8_t)*s++);

        for (int i = 0; i < 4; i++)
            outb(0xE9, (uint8_t)buffer[i]);

        outb(0xE9, '\n');
    } else {
        s = "B2 FAIL\n";
        while (*s)
            outb(0xE9, (uint8_t)*s++);
    }

finish:
    for (;;)
        asm volatile("hlt");
}

#include "io.h"
#include "files.h"

static void print(const char *text)
{
    while (*text != '\0') {
        outb(0xE9, (uint8_t)*text);
        text++;
    }
}

void
__attribute__((noreturn))
__attribute__((section(".start")))
_start(void)
{
    int invalid_fd;
    int shared_fd;
    int written;

    /*
     * Prvi znak imena je cifra, pa ime nije validno.
     * Host treba da vrati -1.
     */
    invalid_fd = open(
        "1invalid.txt",
        FILE_O_WR | FILE_O_CREATE
    );

    if (invalid_fd == -1)
        print("B3 INVALID NAME PASS\n");
    else {
        print("B3 INVALID NAME FAIL\n");
        close(invalid_fd);
    }

    /*
     * a.txt će biti prosleđen hostu preko opcije -f.
     * Na prvom write pozivu host treba da napravi
     * privatnu kopiju za ovu VM.
     */
    shared_fd = open("a.txt", FILE_O_WR);

    if (shared_fd < 0) {
        print("B3 SHARED OPEN FAIL\n");

        for (;;)
            asm volatile("hlt");
    }

    written = write(shared_fd, "COPY", 4);

    if (written == 4)
        print("B3 COPY-ON-WRITE WRITE PASS\n");
    else
        print("B3 COPY-ON-WRITE WRITE FAIL\n");

    if (close(shared_fd) == 0)
        print("B3 CLOSE PASS\n");
    else
        print("B3 CLOSE FAIL\n");

    for (;;)
        asm volatile("hlt");
}
