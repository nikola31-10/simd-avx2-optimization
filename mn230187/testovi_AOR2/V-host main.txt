	size_t memory_size;
	size_t page_size;

	char **guest_paths;
	int guest_count;

	char **shared_file_paths;
	int shared_file_count;

static void print_usage(const char *program_name)
{
	printf(
		"Usage:\n"
		"  %s -m <2|4|8> -p <4|2> "
		"-g <guest1.img> [guest2.img ...] "
		"[-f <file1> [file2 ...]]\n"
		"\n"
		"Options:\n"
		"  -m, --memory   Guest memory size in MB: 2, 4 or 8\n"
		"  -p, --page     Page size: 4 for 4 KB or 2 for 2 MB\n"
		"  -g, --guest    One or more guest image files\n"
		"  -f, --file     Files shared between virtual machines\n",
		program_name
	);
}
static int parse_number(const char *text, unsigned long *value)
{
	char *end;

	*value = strtoul(text, &end, 10);

	if (*text == '\0' || *end != '\0')
		return -1;

	return 0;
}
static int is_option(const char *text)
{
	return strcmp(text, "-m") == 0 ||
	       strcmp(text, "--memory") == 0 ||
	       strcmp(text, "-p") == 0 ||
	       strcmp(text, "--page") == 0 ||
	       strcmp(text, "-g") == 0 ||
	       strcmp(text, "--guest") == 0 ||
	       strcmp(text, "-f") == 0 ||
	       strcmp(text, "--file") == 0;
}
static int parse_arguments(
	int argc,
	char **argv,
	struct program_options *options
)
{
	int i;

	memset(options, 0, sizeof(*options));

	for (i = 1; i < argc; i++) {
		if (strcmp(argv[i], "-m") == 0 ||
		    strcmp(argv[i], "--memory") == 0) {
			unsigned long memory_mb;

			if (i + 1 >= argc ||
			    parse_number(argv[i + 1], &memory_mb) < 0) {
				printf("Error: invalid memory value\n");
				return -1;
			}

			if (memory_mb != 2 &&
			    memory_mb != 4 &&
			    memory_mb != 8) {
				printf("Error: memory must be 2, 4 or 8 MB\n");
				return -1;
			}

			options->memory_size =
				(size_t)memory_mb * 1024u * 1024u;

			i++;
		}
		else if (strcmp(argv[i], "-p") == 0 ||
		         strcmp(argv[i], "--page") == 0) {
			unsigned long page;

			if (i + 1 >= argc ||
			    parse_number(argv[i + 1], &page) < 0) {
				printf("Error: invalid page-size value\n");
				return -1;
			}

			if (page == 4) {
				options->page_size = 4u * 1024u;
			}
			else if (page == 2) {
				options->page_size = 2u * 1024u * 1024u;
			}
			else {
				printf("Error: page size must be 4 KB or 2 MB\n");
				return -1;
			}

			i++;
		}
		else if (strcmp(argv[i], "-g") == 0 ||
		         strcmp(argv[i], "--guest") == 0) {
			int first_guest;
			int count;

			first_guest = i + 1;
			count = 0;

			while (first_guest + count < argc &&
			       !is_option(argv[first_guest + count])) {
				count++;
			}

			if (count == 0) {
				printf("Error: at least one guest image is required\n");
				return -1;
			}

			options->guest_paths = &argv[first_guest];
			options->guest_count = count;

			i += count;
		}
		else if (strcmp(argv[i], "-f") == 0 ||
		         strcmp(argv[i], "--file") == 0) {
			int first_file;
			int count;

			first_file = i + 1;
			count = 0;

			while (first_file + count < argc &&
			       !is_option(argv[first_file + count])) {
				count++;
			}

			if (count == 0) {
				printf("Error: --file requires at least one path\n");
				return -1;
			}

			options->shared_file_paths = &argv[first_file];
			options->shared_file_count = count;

			i += count;
		}
		else {
			printf("Error: unknown option '%s'\n", argv[i]);
			return -1;
		}
	}

	if (options->memory_size == 0) {
		printf("Error: memory option is required\n");
		return -1;
	}

	if (options->page_size == 0) {
		printf("Error: page-size option is required\n");
		return -1;
	}

	if (options->guest_count == 0) {
		printf("Error: at least one guest image is required\n");
		return -1;
	}

	return 0;
}

#define MAX_FILE_NAME 256

enum file_request_state {
	FILE_STATE_WAIT_OPERATION,

	FILE_STATE_OPEN_FLAGS,
	FILE_STATE_OPEN_LENGTH,
	FILE_STATE_OPEN_PATH,
	FILE_STATE_OPEN_RESULT,

	FILE_STATE_CLOSE_FD,
	FILE_STATE_CLOSE_RESULT,

	FILE_STATE_WRITE_FD,
	FILE_STATE_WRITE_COUNT,
	FILE_STATE_WRITE_DATA,
	FILE_STATE_WRITE_RESULT,

	FILE_STATE_READ_FD,
	FILE_STATE_READ_COUNT,
	FILE_STATE_READ_RESULT,
	FILE_STATE_READ_DATA,

	FILE_STATE_LSEEK_FD,
	FILE_STATE_LSEEK_OFFSET,
	FILE_STATE_LSEEK_FLAG,
	FILE_STATE_LSEEK_RESULT
};

#define FILE_BUFFER_SIZE 1024
struct file_request {
	enum file_request_state state;

	int flags;
	int fd;

	uint32_t path_length;
	uint32_t path_received;
	char path[MAX_FILE_NAME];

	int result;

	uint32_t count;
	uint32_t data_received;
	char buffer[FILE_BUFFER_SIZE];

	int offset;
};
#define MAX_OPEN_FILES 32

	int used;
	int host_fd;
	int flags;

	int is_shared;
	int copied;

	char path[MAX_FILE_NAME];
	char original_path[MAX_FILE_NAME];
};
static int find_free_guest_fd(struct vm_file_entry *open_files)
{
	for (int i = 0; i < MAX_OPEN_FILES; i++) {
		if (!open_files[i].used)
			return i;
	}

	return -1;
}

	char path[MAX_FILE_NAME];
};
static int is_valid_file_name(const char *path)
{
	int i;

	if (path == NULL || path[0] == '\0')
		return 0;

	/* Ime mora da počne slovom. */
	if (!((path[0] >= 'a' && path[0] <= 'z') ||
	      (path[0] >= 'A' && path[0] <= 'Z'))) {
		return 0;
	}

	/* Ostali znakovi mogu biti slova, cifre ili tačka. */
	for (i = 1; path[i] != '\0'; i++) {
		char c = path[i];

		if (!((c >= 'a' && c <= 'z') ||
		      (c >= 'A' && c <= 'Z') ||
		      (c >= '0' && c <= '9') ||
		      c == '.')) {
			return 0;
		}
	}

	return 1;
}
static int is_shared_file(
	const char *path,
	char **shared_file_paths,
	int shared_file_count
)
{
	int i;

	for (i = 0; i < shared_file_count; i++) {
		if (strcmp(path, shared_file_paths[i]) == 0)
			return 1;
	}

	return 0;
}
static int copy_file(const char *source_path, const char *destination_path)
{
	int source_fd;
	int destination_fd;
	char buffer[1024];
	ssize_t bytes_read;

	source_fd = open(source_path, O_RDONLY);
	if (source_fd < 0) {
		perror("open source file");
		return -1;
	}

	destination_fd = open(
		destination_path,
		O_WRONLY | O_CREAT | O_TRUNC,
		0644
	);

	if (destination_fd < 0) {
		perror("open destination file");
		close(source_fd);
		return -1;
	}

	while ((bytes_read = read(
		    source_fd,
		    buffer,
		    sizeof(buffer)
	    )) > 0) {
		ssize_t total_written = 0;

		while (total_written < bytes_read) {
			ssize_t written = write(
				destination_fd,
				buffer + total_written,
				(size_t)(bytes_read - total_written)
			);

			if (written < 0) {
				perror("write copied file");
				close(source_fd);
				close(destination_fd);
				return -1;
			}

			total_written += written;
		}
	}

	if (bytes_read < 0) {
		perror("read source file");
		close(source_fd);
		close(destination_fd);
		return -1;
	}

	close(source_fd);
	close(destination_fd);

	return 0;
}
static int ensure_private_copy(
	struct vm_file_entry *entry,
	int vm_id
)
{
	char copy_path[MAX_FILE_NAME];
	off_t current_offset;
	int host_flags;
	int new_fd;

	if (!entry->is_shared || entry->copied)
		return 0;

	current_offset = lseek(entry->host_fd, 0, SEEK_CUR);
	if (current_offset < 0)
		return -1;

	if (snprintf(
		    copy_path,
		    sizeof(copy_path),
		    "vm%d_%s",
		    vm_id,
		    entry->original_path
	    ) >= (int)sizeof(copy_path)) {
		fprintf(stderr, "Copy file name is too long\n");
		return -1;
	}

	if (copy_file(entry->original_path, copy_path) < 0)
		return -1;

	host_flags = 0;

	if (entry->flags & FILE_O_RDWR)
		host_flags |= O_RDWR;
	else if (entry->flags & FILE_O_WR)
		host_flags |= O_WRONLY;
	else
		host_flags |= O_RDONLY;

	close(entry->host_fd);

	new_fd = open(copy_path, host_flags);
	if (new_fd < 0) {
		perror("open private copy");
		entry->host_fd = -1;
		return -1;
	}

	if (lseek(new_fd, current_offset, SEEK_SET) < 0) {
		perror("seek private copy");
		close(new_fd);
		entry->host_fd = -1;
		return -1;
	}

	entry->host_fd = new_fd;
	entry->copied = 1;

	strncpy(entry->path, copy_path, MAX_FILE_NAME - 1);
	entry->path[MAX_FILE_NAME - 1] = '\0';

	printf(
		"VM %d: copy-on-write %s -> %s\n",
		vm_id,
		entry->original_path,
		entry->path
	);

	return 0;
}
struct shared_buffer {
	pthread_mutex_t mutex;

	uint8_t data[BUFFER_SIZE];
	uint32_t size;

	int writer_finished;
	int readers_finished;
	int reader_count;

	uint32_t generation;
	int last_block;
};
static int run_vm(
	int vm_id,
	const char *guest_path,
	size_t memory_size,
	size_t page_size,
	char **shared_file_paths,
	int shared_file_count,
	struct shared_buffer *shared_buffer
)
{
	struct vm_file_entry open_files[MAX_OPEN_FILES];
	struct file_request file_request;
	struct vm v;
	struct kvm_sregs sregs;
	struct kvm_regs regs;
	int stop = 0;
	int ret;
	int irq_pending = 1;
	
	int vm_role;
	int role_delivered = 0;

	uint32_t shared_write_expected = 0;
	uint32_t shared_write_received = 0;
	int shared_write_active = 0;
	int shared_ack_ready = 0;

	uint32_t shared_read_position = 0;
	int shared_read_active = 0;

	int writer_waiting_for_readers = 0;

	//int reader_ack_sent = 0;

	int transfer_finished = 0;

	uint32_t last_generation_read = 0;

	if (vm_id == 0)
		vm_role = ROLE_WRITER;
	else
		vm_role = ROLE_READER;
	//(void)page_size;

	memset(&file_request, 0, sizeof(file_request));
	file_request.state = FILE_STATE_WAIT_OPERATION;

	memset(open_files, 0, sizeof(open_files));

	for (int i = 0; i < MAX_OPEN_FILES; i++)
		open_files[i].host_fd = -1;

	if (vm_init(&v, memory_size)) {
		printf("Failed to init the VM for guest: %s\n", guest_path);
		return 1;
	}

	if (ioctl(v.vcpu_fd, KVM_GET_SREGS, &sregs) < 0) {
		perror("KVM_GET_SREGS");
		vm_destroy(&v);
		return 1;
	}

	setup_long_mode(&v, &sregs, page_size);

	if (ioctl(v.vcpu_fd, KVM_SET_SREGS, &sregs) < 0) {
		perror("KVM_SET_SREGS");
		vm_destroy(&v);
		return 1;
	}

	uint64_t guest_load_addr;

	if (page_size == 2u * 1024u * 1024u)
		guest_load_addr = 0;
	else
		guest_load_addr = GUEST_START_ADDR;
	if (load_guest_image(&v, guest_path, guest_load_addr) < 0) {
		printf("Failed to load guest image: %s\n", guest_path);
		vm_destroy(&v);
		return 1;
	}

	memset(&regs, 0, sizeof(regs));
	regs.rflags = 0x2;
	regs.rip = 0;

	if (page_size == 2u * 1024u * 1024u)
		regs.rsp = 0x1F0000;
	else
		regs.rsp = 2u << 20;

	if (ioctl(v.vcpu_fd, KVM_SET_REGS, &regs) < 0) {
		perror("KVM_SET_REGS");
		vm_destroy(&v);
		return 1;
	}

	v.run->request_interrupt_window = irq_pending > 0;

	while (!stop) {
		ret = ioctl(v.vcpu_fd, KVM_RUN, 0);

		if (ret == -1) {
			perror("KVM_RUN");
			vm_destroy(&v);
			return 1;
		}

		switch (v.run->exit_reason) {
		case KVM_EXIT_IO:
			{
				char *data = (char *)v.run + v.run->io.data_offset;

				if (v.run->io.port == SHARED_IO_PORT) {
					/* Prvo IN čitanje služi za dodelu uloge. */
					if (
						v.run->io.direction == KVM_EXIT_IO_IN &&
						v.run->io.size == 4 &&
						!role_delivered
					) {
						*(uint32_t *)data = (uint32_t)vm_role;
						role_delivered = 1;

						/*
						* Zakazujemo naredni IRQ koji će pokrenuti
						* prvu stvarnu rundu prenosa.
						*/
						if (vm_role == ROLE_WRITER) {
							irq_pending++;
							v.run->request_interrupt_window = 1;
						}

						break;
					}

					/* Samo writer sme da šalje podatke. */
					if (vm_role == ROLE_WRITER &&
						v.run->io.direction == KVM_EXIT_IO_OUT) {

						/* Writer prvo šalje broj bajtova kao 32-bitnu vrednost. */
						if (v.run->io.size == 4 && !shared_write_active) {
							uint32_t requested = *(uint32_t *)data;

							if (requested > BUFFER_SIZE) {
								fprintf(
									stderr,
									"VM %d: shared message too large: %u\n",
									vm_id,
									requested
								);

								stop = 1;
								break;
							}

							pthread_mutex_lock(&shared_buffer->mutex);

							shared_buffer->size = 0;
							shared_buffer->writer_finished = 0;

							pthread_mutex_unlock(&shared_buffer->mutex);

							shared_write_expected = requested;
							shared_write_received = 0;
							shared_write_active = requested > 0;

							if (requested == 0) {
								pthread_mutex_lock(&shared_buffer->mutex);

								shared_buffer->size = 0;
								shared_buffer->writer_finished = 1;
								shared_buffer->readers_finished = 0;
								shared_buffer->last_block = 1;
								shared_buffer->generation++;

								pthread_mutex_unlock(&shared_buffer->mutex);

								shared_ack_ready = 1;
								writer_waiting_for_readers = 1;
							}

							break;
						}

						/* Nakon dužine writer šalje bajt po bajt. */
						if (v.run->io.size == 1 && shared_write_active) {
							pthread_mutex_lock(&shared_buffer->mutex);

							shared_buffer->data[shared_write_received] =
								(uint8_t)*data;

							pthread_mutex_unlock(&shared_buffer->mutex);

							shared_write_received++;

							if (shared_write_received == shared_write_expected) {
								pthread_mutex_lock(&shared_buffer->mutex);

								shared_buffer->size = shared_write_received;
								shared_buffer->writer_finished = 1;
								shared_buffer->readers_finished = 0;
								shared_buffer->last_block =
									(shared_write_received < BUFFER_SIZE);

								shared_buffer->generation++;

								pthread_mutex_unlock(&shared_buffer->mutex);
								shared_write_active = 0;
								shared_ack_ready = 1;
								writer_waiting_for_readers = 1;

								printf(
									"VM %d wrote %u bytes to shared buffer\n",
									vm_id,
									shared_write_received
								);
							}

							break;
								}
						}
					/* Reader čita iz zajedničkog bafera */
						if (
							vm_role == ROLE_READER &&
							v.run->io.direction == KVM_EXIT_IO_IN
						) {
							pthread_mutex_lock(&shared_buffer->mutex);

							if (!shared_buffer->writer_finished) {
								pthread_mutex_unlock(&shared_buffer->mutex);

								*(uint32_t *)data = SHARED_NOT_READY;
								break;
							}

							if (v.run->io.size == 4 && !shared_read_active) {

								*(uint32_t *)data = shared_buffer->size;

								shared_read_position = 0;
								shared_read_active = 1;

								pthread_mutex_unlock(&shared_buffer->mutex);
								break;
							}

							if (v.run->io.size == 1 && shared_read_active) {

								*data = shared_buffer->data[shared_read_position++];

								if (shared_read_position == shared_buffer->size)
									shared_read_active = 0;

								pthread_mutex_unlock(&shared_buffer->mutex);
								break;
							}

							pthread_mutex_unlock(&shared_buffer->mutex);
						}
						

					fprintf(
						stderr,
						"VM %d: invalid access to shared port 0x%x\n",
						vm_id,
						SHARED_IO_PORT
					);

					stop = 1;
					break;
				}

				if (v.run->io.port == SHARED_ACK_PORT) {
					if (
						vm_role == ROLE_WRITER &&
						v.run->io.direction == KVM_EXIT_IO_IN &&
						v.run->io.size == 4 &&
						shared_ack_ready
					) {
						uint32_t acknowledged;
						int is_last_block;

						pthread_mutex_lock(&shared_buffer->mutex);

						if (
							writer_waiting_for_readers &&
							shared_buffer->readers_finished !=
							shared_buffer->reader_count
						) {
							pthread_mutex_unlock(&shared_buffer->mutex);

							*(uint32_t *)data = SHARED_NOT_READY;
							break;
						}

						acknowledged = shared_write_received;
						is_last_block = shared_buffer->last_block;

						pthread_mutex_unlock(&shared_buffer->mutex);

						*(uint32_t *)data = acknowledged;

						writer_waiting_for_readers = 0;
						shared_ack_ready = 0;

						if (is_last_block) {
							transfer_finished = 1;
						} else {
							/*
							* Zakazujemo writeru sledeći IRQ,
							* odnosno sledeću rundu čitanja fajla.
							*/
							irq_pending++;
							v.run->request_interrupt_window = 1;

							pthread_mutex_lock(&shared_buffer->mutex);

							shared_buffer->writer_finished = 0;
							shared_buffer->size = 0;

							pthread_mutex_unlock(&shared_buffer->mutex);
						}

						break;
					}
					if (
						vm_role == ROLE_READER &&
						v.run->io.direction == KVM_EXIT_IO_OUT &&
						v.run->io.size == 4
					) {
						uint32_t acknowledged = *(uint32_t *)data;

						pthread_mutex_lock(&shared_buffer->mutex);

						if (last_generation_read == shared_buffer->generation) {
							fprintf(
								stderr,
								"VM %d sent duplicate ACK for generation %u\n",
								vm_id,
								shared_buffer->generation
							);

							pthread_mutex_unlock(&shared_buffer->mutex);
							stop = 1;
							break;
						}

						if (acknowledged != shared_buffer->size) {
							fprintf(
								stderr,
								"VM %d acknowledged %u bytes, expected %u\n",
								vm_id,
								acknowledged,
								shared_buffer->size
							);

							pthread_mutex_unlock(&shared_buffer->mutex);

							stop = 1;
							break;
						}

						//reader_ack_sent = 1;
						shared_buffer->readers_finished++;
						last_generation_read = shared_buffer->generation;
						

						printf(
							"VM %d finished reading %u bytes (%d/%d)\n",
							vm_id,
							acknowledged,
							shared_buffer->readers_finished,
							shared_buffer->reader_count
						);
						if (shared_buffer->last_block)
    						transfer_finished = 1;
							
						pthread_mutex_unlock(&shared_buffer->mutex);

						break;
					}

					fprintf(
						stderr,
						"VM %d: invalid access to ACK port 0x%x\n",
						vm_id,
						SHARED_ACK_PORT
					);

					stop = 1;
					break;
				}

				/* Obrada fajl protokola preko porta 0x0278 */
				if (v.run->io.port == FILE_IO_PORT) {
					if (v.run->io.direction == KVM_EXIT_IO_OUT) {
						if (v.run->io.size == 4) {
							uint32_t value = *(uint32_t *)data;

							switch (file_request.state) {
							case FILE_STATE_WAIT_OPERATION:
								if (value == FILE_OP_OPEN) {
									file_request.state = FILE_STATE_OPEN_FLAGS;
								}
								else if (value == FILE_OP_CLOSE) {
									file_request.state = FILE_STATE_CLOSE_FD;
								}
								else if (value == FILE_OP_WRITE) {
									file_request.state = FILE_STATE_WRITE_FD;
								}
								else if (value == FILE_OP_READ) {
									file_request.state = FILE_STATE_READ_FD;
								}
								else if (value == FILE_OP_LSEEK) {
									file_request.state = FILE_STATE_LSEEK_FD;
								}
								else {
									fprintf(
										stderr,
										"Unsupported file operation: %u\n",
										value
									);
									stop = 1;
								}

								break;

							case FILE_STATE_OPEN_FLAGS:
								file_request.flags = (int)value;
								file_request.state = FILE_STATE_OPEN_LENGTH;
								break;

							case FILE_STATE_OPEN_LENGTH:
								if (value == 0 || value >= MAX_FILE_NAME) {
									file_request.result = -1;
									file_request.state = FILE_STATE_OPEN_RESULT;
									break;
								}

								file_request.path_length = value;
								file_request.path_received = 0;
								file_request.state = FILE_STATE_OPEN_PATH;
								break;
							case FILE_STATE_CLOSE_FD:
								file_request.fd = (int32_t)value;

								if (
									file_request.fd < 0 ||
									file_request.fd >= MAX_OPEN_FILES ||
									!open_files[file_request.fd].used
								) {
									file_request.result = -1;
								} else {
									if (close(open_files[file_request.fd].host_fd) < 0) {
										file_request.result = -1;
									} else {
										open_files[file_request.fd].used = 0;
										open_files[file_request.fd].host_fd = -1;
										open_files[file_request.fd].flags = 0;
										open_files[file_request.fd].path[0] = '\0';
										open_files[file_request.fd].is_shared = 0;
										open_files[file_request.fd].copied = 0;
										open_files[file_request.fd].original_path[0] = '\0';

										file_request.result = 0;
									}
								}

								file_request.state = FILE_STATE_CLOSE_RESULT;
								break;
							

							case FILE_STATE_WRITE_FD:
								file_request.fd = (int32_t)value;
								file_request.state = FILE_STATE_WRITE_COUNT;
								break;

							case FILE_STATE_WRITE_COUNT:
								if (
									file_request.fd < 0 ||
									file_request.fd >= MAX_OPEN_FILES ||
									!open_files[file_request.fd].used ||
									value > FILE_BUFFER_SIZE
								) {
									file_request.result = -1;
									file_request.state = FILE_STATE_WRITE_RESULT;
									break;
								}

								file_request.count = value;
								file_request.data_received = 0;

								if (file_request.count == 0) {
									file_request.result = 0;
									file_request.state = FILE_STATE_WRITE_RESULT;
								} else {
									file_request.state = FILE_STATE_WRITE_DATA;
								}
								break;
							
							case FILE_STATE_READ_FD:
								file_request.fd = (int32_t)value;
								file_request.state = FILE_STATE_READ_COUNT;
								break;

							case FILE_STATE_READ_COUNT:
								if (
									file_request.fd < 0 ||
									file_request.fd >= MAX_OPEN_FILES ||
									!open_files[file_request.fd].used ||
									value > FILE_BUFFER_SIZE
								) {
									file_request.result = -1;
									file_request.state = FILE_STATE_READ_RESULT;
									break;
								}

								file_request.count = value;

								if (file_request.count == 0) {
									file_request.result = 0;
									file_request.state = FILE_STATE_READ_RESULT;
								} else {
									ssize_t bytes_read;

									bytes_read = read(
										open_files[file_request.fd].host_fd,
										file_request.buffer,
										file_request.count
									);

									if (bytes_read < 0)
										file_request.result = -1;
									else
										file_request.result = (int)bytes_read;

									file_request.data_received = 0;
									file_request.state = FILE_STATE_READ_RESULT;
								}
								break;
							
							case FILE_STATE_LSEEK_FD:
								file_request.fd = (int32_t)value;
								file_request.state = FILE_STATE_LSEEK_OFFSET;
								break;

							case FILE_STATE_LSEEK_OFFSET:
								file_request.offset = (int32_t)value;
								file_request.state = FILE_STATE_LSEEK_FLAG;
								break;

							case FILE_STATE_LSEEK_FLAG:
								if (
									file_request.fd < 0 ||
									file_request.fd >= MAX_OPEN_FILES ||
									!open_files[file_request.fd].used
								) {
									file_request.result = -1;
									file_request.state = FILE_STATE_LSEEK_RESULT;
									break;
								}

								if (value == FILE_SEEK_SET) {
									file_request.result = (int)lseek(
										open_files[file_request.fd].host_fd,
										file_request.offset,
										SEEK_SET
									);
								}
								else if (value == FILE_SEEK_END) {
									file_request.result = (int)lseek(
										open_files[file_request.fd].host_fd,
										file_request.offset,
										SEEK_END
									);
								}
								else {
									file_request.result = -1;
								}

								file_request.state = FILE_STATE_LSEEK_RESULT;
								break;
							default:
								fprintf(stderr, "Unexpected 32-bit file output\n");
								stop = 1;
								break;
							}
						}
						//ovde
						else if (v.run->io.size == 1) {
							if (file_request.state == FILE_STATE_OPEN_PATH) {
								file_request.path[file_request.path_received++] = *data;

								if (
									file_request.path_received ==
									file_request.path_length
								) {
									int host_flags = 0;
									int guest_fd;
									int host_fd;

									file_request.path[file_request.path_received] = '\0';

									if (!is_valid_file_name(file_request.path)) {
										fprintf(
											stderr,
											"Invalid file name: %s\n",
											file_request.path
										);

										file_request.result = -1;
										file_request.state = FILE_STATE_OPEN_RESULT;
										break;
									}
									if (file_request.flags & FILE_O_RDWR)
										host_flags |= O_RDWR;
									else if (file_request.flags & FILE_O_WR)
										host_flags |= O_WRONLY;
									else
										host_flags |= O_RDONLY;

									if (file_request.flags & FILE_O_CREATE)
										host_flags |= O_CREAT;

									guest_fd = find_free_guest_fd(open_files);

									if (guest_fd < 0) {
										file_request.result = -1;
									} else {
										host_fd = open(
											file_request.path,
											host_flags,
											0644
										);

										if (host_fd < 0) {
											file_request.result = -1;
										} else {
											open_files[guest_fd].used = 1;
											open_files[guest_fd].host_fd = host_fd;
											open_files[guest_fd].flags =
												file_request.flags;

											open_files[guest_fd].is_shared = is_shared_file(
												file_request.path,
												shared_file_paths,
												shared_file_count
											);

											open_files[guest_fd].copied = 0;

											strncpy(
												open_files[guest_fd].original_path,
												file_request.path,
												MAX_FILE_NAME - 1
											);

											open_files[guest_fd]
												.original_path[MAX_FILE_NAME - 1] = '\0';

											strncpy(
												open_files[guest_fd].path,
												file_request.path,
												MAX_FILE_NAME - 1
											);

											open_files[guest_fd]
												.path[MAX_FILE_NAME - 1] = '\0';

											file_request.result = guest_fd;
										}
									}

									file_request.state = FILE_STATE_OPEN_RESULT;
								}
							}
							else if (file_request.state == FILE_STATE_WRITE_DATA) {
								file_request.buffer[file_request.data_received++] = *data;

								if (
									file_request.data_received ==
									file_request.count
								) {
									ssize_t written;

									if (ensure_private_copy(
											&open_files[file_request.fd],
											vm_id
										) < 0) {
										file_request.result = -1;
										file_request.state = FILE_STATE_WRITE_RESULT;
										break;
									}

									written = write(
										open_files[file_request.fd].host_fd,
										file_request.buffer,
										file_request.count
									);

									if (written < 0)
										file_request.result = -1;
									else
										file_request.result = (int)written;

									file_request.state = FILE_STATE_WRITE_RESULT;
								}
							}
							else {
								fprintf(stderr, "Unexpected byte in file protocol\n");
								stop = 1;
							}

							break;
						}///OVOOOO
						else {
							fprintf(stderr, "Invalid file-protocol output\n");
							stop = 1;
						}

						break;
					}

					if (
						v.run->io.direction == KVM_EXIT_IO_IN &&
						v.run->io.size == 4 &&
						(
							file_request.state == FILE_STATE_OPEN_RESULT ||
							file_request.state == FILE_STATE_CLOSE_RESULT ||
							file_request.state == FILE_STATE_WRITE_RESULT ||
							file_request.state == FILE_STATE_READ_RESULT ||
							file_request.state == FILE_STATE_LSEEK_RESULT
						)
					) {
						*(int32_t *)data = file_request.result;

						if (
							file_request.state == FILE_STATE_READ_RESULT &&
							file_request.result > 0
						) {
							file_request.state = FILE_STATE_READ_DATA;
							file_request.data_received = 0;
						} else {
							file_request.state = FILE_STATE_WAIT_OPERATION;
							file_request.path_length = 0;
							file_request.path_received = 0;
							file_request.fd = -1;
							file_request.count = 0;
							file_request.data_received = 0;
							file_request.offset = 0;
						}

						break;
					}
					if (
						v.run->io.direction == KVM_EXIT_IO_IN &&
						v.run->io.size == 1 &&
						file_request.state == FILE_STATE_READ_DATA
					) {
						*data = file_request.buffer[file_request.data_received++];

						if (file_request.data_received == (uint32_t)file_request.result) {
							file_request.state = FILE_STATE_WAIT_OPERATION;
							file_request.fd = -1;
							file_request.count = 0;
							file_request.data_received = 0;
						}

						break;
					}

					fprintf(stderr, "Invalid file-protocol input\n");
					stop = 1;
					break;
				}

				/* Obrada serijskog porta 0xE9 */
				if (v.run->io.port == 0xE9 && v.run->io.size == 1) {
					if (v.run->io.direction == KVM_EXIT_IO_OUT) {
						printf("%c", *data);
						fflush(stdout);
					}
					else if (v.run->io.direction == KVM_EXIT_IO_IN) {
						int input = getchar();

						if (input == EOF)
							*data = 0;
						else
							*data = (char)input;
					}

					break;
				}

				printf(
					"Unexpected I/O operation for %s: port=0x%x, size=%u\n",
					guest_path,
					v.run->io.port,
					v.run->io.size
				);
				stop = 1;
				break;
			}
		case KVM_EXIT_IRQ_WINDOW_OPEN:
			if (irq_pending > 0) {
				if (inject_irq(&v, SHARED_IRQ) < 0) {
					vm_destroy(&v);
					return 1;
				}

				irq_pending--;
			} else {
				v.run->request_interrupt_window = 0;
			}
			break;

		case KVM_EXIT_HLT:
			/* Već zakazan IRQ, najčešće za writer. */
			if (irq_pending > 0) {
				if (inject_irq(&v, SHARED_IRQ) < 0) {
					stop = 1;
					break;
				}

				irq_pending--;
				v.run->request_interrupt_window = 0;
				break;
			}

			if (transfer_finished) {
				printf("KVM_EXIT_HLT for %s\n", guest_path);
				stop = 1;
				break;
			}

			/*
			* Reader dobija IRQ tek kada postoji nova,
			* još nepročitana generacija bafera.
			*/
			if (role_delivered && vm_role == ROLE_READER) {
				int round_ready;

				pthread_mutex_lock(&shared_buffer->mutex);

				round_ready =
					shared_buffer->writer_finished &&
					shared_buffer->generation >
						last_generation_read;

				pthread_mutex_unlock(&shared_buffer->mutex);

				if (round_ready) {
					if (inject_irq(&v, SHARED_IRQ) < 0) {
						stop = 1;
						break;
					}

					break;
				}
			}

			/*
			* Nema spremnog rada. Sprečavamo agresivno
			* zauzimanje procesora dok čekamo drugu VM.
			*/
			usleep(1000);
			break;

		case KVM_EXIT_SHUTDOWN:
			printf("VM shutdown for %s\n", guest_path);
			stop = 1;
			break;

		default:
			printf(
				"Unexpected VM exit for %s: reason=%d\n",
				guest_path,
				v.run->exit_reason
			);
			stop = 1;
			break;
		}
	}

	for (int i = 0; i < MAX_OPEN_FILES; i++) {
		if (open_files[i].used) {
			close(open_files[i].host_fd);
			open_files[i].used = 0;
			open_files[i].host_fd = -1;
		}
	}

	vm_destroy(&v);
	return 0;
}
struct vm_thread_args {
	int vm_id;
    const char *guest_path;
    size_t memory_size;
    size_t page_size;

    char **shared_file_paths;
    int shared_file_count;

    int result;

	struct shared_buffer *shared_buffer;
};
static void *vm_thread_main(void *arg)
{
	struct vm_thread_args *args = arg;

	args->result = run_vm(
		args->vm_id,
		args->guest_path,
		args->memory_size,
		args->page_size,
		args->shared_file_paths,
		args->shared_file_count,
		args->shared_buffer
	);

	return NULL;
}
int main(int argc, char *argv[])
{
	struct program_options options;
	struct vm_thread_args *thread_args;
	pthread_t *threads;
	struct shared_buffer shared_buffer;
	int i;
	int exit_status = 0;

	if (parse_arguments(argc, argv, &options) < 0) {
		print_usage(argv[0]);
		return 1;
	}
	memset(&shared_buffer, 0, sizeof(shared_buffer));

	if (pthread_mutex_init(&shared_buffer.mutex, NULL) != 0) {
		fprintf(stderr, "Failed to initialize shared-buffer mutex\n");
		return 1;
	}

	printf("Guests:\n");

	for (i = 0; i < options.guest_count; i++)
		printf("  %s\n", options.guest_paths[i]);

	printf("Shared files:\n");

	for (i = 0; i < options.shared_file_count; i++)
    	printf("  %s\n", options.shared_file_paths[i]);

	threads = calloc((size_t)options.guest_count, sizeof(*threads));
	thread_args = calloc(
		(size_t)options.guest_count,
		sizeof(*thread_args)
	);

	if (!threads || !thread_args) {
		perror("calloc");
		free(threads);
		free(thread_args);
		return 1;
	}
	shared_buffer.reader_count = options.guest_count > 0 ? options.guest_count - 1 : 0;
	for (i = 0; i < options.guest_count; i++) {
		thread_args[i].guest_path = options.guest_paths[i];
		thread_args[i].memory_size = options.memory_size;
		thread_args[i].page_size = options.page_size;
		thread_args[i].result = 1;
		thread_args[i].shared_file_paths = options.shared_file_paths;
		thread_args[i].shared_file_count = options.shared_file_count;
		thread_args[i].vm_id = i;
		thread_args[i].shared_buffer = &shared_buffer;

		
		if (pthread_create(
			    &threads[i],
			    NULL,
			    vm_thread_main,
			    &thread_args[i]
		    ) != 0) {
			fprintf(
				stderr,
				"Failed to create thread for guest: %s\n",
				options.guest_paths[i]
			);

			options.guest_count = i;
			exit_status = 1;
			break;
		}
	}

	for (i = 0; i < options.guest_count; i++) {
		if (pthread_join(threads[i], NULL) != 0) {
			fprintf(stderr, "Failed to join VM thread %d\n", i);
			exit_status = 1;
		}

		if (thread_args[i].result != 0)
			exit_status = 1;
	}

	free(threads);
	free(thread_args);

	pthread_mutex_destroy(&shared_buffer.mutex);

	return exit_status;
}
